## 2. Container View (C4Container)

```mermaid
C4Container
  title Container Diagram: Tools Core Components

  Person(user, "Engineer / Developer", "User of desktop GUIs or consumer of Python packages.")

  Container_Boundary(c1, "Tools Platform") {
    Container(launchers, "Unified & CLI Launchers", "Python 3.11+, PyQt6", "UnifiedToolsLauncher desktop hub and launch.py CLI auto-discovering all tools via gui_registration.")
    Container(process_calcs, "Process & Engineering Calculators", "Python 3.11+, NumPy, SciPy", "Standalone chemical, thermodynamic, and mechanical calculators (flare, scrubber, dewpoint, pressure drop, vessel drafter).")
    Container(robotics_engine, "Robotics & Kinematics Engine", "Python 3.11+, NumPy, Trimesh, Rust", "URDF/MJCF builders, humanoid parametric modelers, inertia calculators, and rotation converters (SE3, twists, quaternions).")
    Container(signal_studio, "Signal Toolkit & Studio", "Python 3.11+, NumPy, SciPy, PyQt6", "Waveform generation, digital filtering (Butterworth, Chebyshev), curve fitting, and time-series signal analysis.")
    Container(mocap_core, "Markerless Mocap Contracts & Lab", "Python 3.11+, NumPy, OpenCV", "Vendor-neutral camera acquisition protocol, C3D exchange, calibration (intrinsic/extrinsic), and reference CLI/services.")
    Container(swing_sim_lib, "Swing Sim & Biomechanics Library", "Python 3.11+, NumPy, Rust core", "Impact mechanics, rate of closure, force attribution, localized torque profiles, and delivery interchange contracts.")
    Container(scada_p1am, "SCADA & Industrial Control", "Python 3.11+, FastAPI, PyModbus", "P1AM-100 PLC HMI, Modbus communication, safety interlocks, historian, and web/desktop telemetry.")
  }

  System_Ext(upstream_drift, "UpstreamDrift", "Consumes swing_sim, mocap protocols, and signal toolkits.")
  System_Ext(gasification_model, "Gasification_Model", "Consumes process and thermodynamic calculators.")

  Rel(user, launchers, "Selects and opens tools via", "PyQt6 GUI / CLI")
  Rel(launchers, process_calcs, "Launches", "Python Process")
  Rel(launchers, robotics_engine, "Launches", "Python Process")
  Rel(launchers, signal_studio, "Launches", "Python Process")
  Rel(launchers, mocap_core, "Orchestrates", "Python Process / CLI")
  Rel(launchers, swing_sim_lib, "Launches", "Python Process / Web")
  Rel(launchers, scada_p1am, "Launches", "Web / FastAPI")

  Rel(upstream_drift, swing_sim_lib, "Imports impact models and trajectory contracts from", "Python Package")
  Rel(upstream_drift, mocap_core, "Adopts C3D serialization and camera acquisition schemas from", "Python Package")
  Rel(gasification_model, process_calcs, "Imports thermodynamic and process calculators from", "Python Package")
```
